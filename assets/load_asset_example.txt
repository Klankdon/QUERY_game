# Inside QueryGameEngine.__init__ (after pygame.init())
        self.load_assets()

    def load_assets(self):
        """Loads and scales pixel art or chibi sprite graphics."""
        # Fallback to colored rects if image files are missing during dev
        self.assets = {}
        try:
            self.player_img = pygame.image.load("assets/player_chibi.png").convert_alpha()
            self.player_img = pygame.transform.scale(self.player_img, (32, 32))
            
            self.mob_img = pygame.image.load("assets/daemon_sprite.png").convert_alpha()
            self.mob_img = pygame.transform.scale(self.mob_img, (32, 32))
        except pygame.error:
            # If files don't exist yet, we can fall back or draw placeholders
            self.player_img = None
            self.mob_img = None
